def make_variant(amount, count, tag):
    doc = pymupdf.open("build/blank_form.pdf")
    pg = doc[0]
    rects = [pymupdf.Rect(288.0, 756.1, 306.0, 768.1),   # Historical private commentary omitted; executable code retained.
             pymupdf.Rect(408.1, 756.1, 414.1, 768.1)]   # Historical private commentary omitted; executable code retained.
    for r in rects:
        pg.add_redact_annot(r)
    pg.apply_redactions(images=pymupdf.PDF_REDACT_IMAGE_NONE,
                        graphics=pymupdf.PDF_REDACT_LINE_ART_NONE,
                        text=pymupdf.PDF_REDACT_TEXT_REMOVE)
    pg.insert_text((288.0, 766.4), str(amount), fontname="helv", fontsize=12, color=(0, 0, 0))
    pg.insert_text((408.1, 766.4), str(count), fontname="helv", fontsize=12, color=(0, 0, 0))
    p = f"build/form_{tag}.pdf"
    doc.save(p)
    doc.close()
    return p

def img_bytes(doc, xref):
    return doc.extract_image(xref)["image"]

GAP_X = (BX1 - BX0 - 2 * IW) / 3.0        # Historical private commentary omitted; executable code retained.
GAP_Y = 60.0

def build_page(variant_pdf, items):
    doc = pymupdf.open(variant_pdf)
    page = out.new_page(width=W, height=H)
    page.show_pdf_page(page.rect, doc, 0)
    doc.close()
    rows = [items[i:i + 2] for i in range(0, len(items), 2)]
    block_h = len(rows) * IH + (len(rows) - 1) * GAP_Y
    top = BY0 + (BY1 - BY0 - block_h) / 2.0
    for ri, row in enumerate(rows):
        y = top + ri * (IH + GAP_Y)
        row_w = len(row) * IW + (len(row) - 1) * GAP_X
        left = BX0 + (BX1 - BX0 - row_w) / 2.0
        for ci, (_, _, xref, _r) in enumerate(row):
            x = left + ci * (IW + GAP_X)
            page.insert_image(pymupdf.Rect(x, y, x + IW, y + IH),
                              stream=img_bytes(src, xref))
    return page
