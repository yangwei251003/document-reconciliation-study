# -*- coding: utf-8 -*-
import pymupdf, os, re, glob
from rapidocr_onnxruntime import RapidOCR

SRC_PDF = '<LOCAL_INPUT_OR_OUTPUT_PATH>'
IMG_DIR = "build/wz"
BACKUP  = "build/old_pdf"
os.makedirs(BACKUP, exist_ok=True)

ocr = RapidOCR()
PAGE_W = 595.0
os.makedirs("build/new_pdf", exist_ok=True)

def upscale_png(path, factor=2):
    pm = pymupdf.Pixmap(path)
    doc = pymupdf.open()
    page = doc.new_page(width=pm.width, height=pm.height)
    page.insert_image(pymupdf.Rect(0, 0, pm.width, pm.height), filename=path)
    pix = page.get_pixmap(matrix=pymupdf.Matrix(factor, factor))
    data = pix.tobytes("png")
    doc.close()
    return data, pm.width * factor, pm.height * factor

imgs = sorted(glob.glob(os.path.join(IMG_DIR, "*.jpeg")),
              key=lambda p: int(re.match(r"(\d+)", os.path.basename(p)).group(1)))

for img in imgs:
    base = os.path.splitext(os.path.basename(img))[0]
    final_pdf = os.path.join(SRC_PDF, base + ".pdf")
    out_pdf = os.path.join("build/new_pdf", base + ".pdf")

    # Historical private commentary omitted; executable code retained.
    if os.path.exists(final_pdf):
        bak = os.path.join(BACKUP, base + ".pdf")
        if not os.path.exists(bak):
            import shutil; shutil.copyfile(final_pdf, bak)

    res, _ = ocr(img)
    res = res or []

    pm0 = pymupdf.Pixmap(img)
    page_h = PAGE_W * pm0.height / pm0.width
    scale = PAGE_W / pm0.width

    png, iw, ih = upscale_png(img, 2)

    doc = pymupdf.open()
    page = doc.new_page(width=PAGE_W, height=page_h)
    page.insert_image(pymupdf.Rect(0, 0, PAGE_W, page_h), stream=png)
    page.insert_image  # Historical private commentary omitted; executable code retained.

    n = 0
    for item in res:
        box, txt = item[0], item[1]
        conf = float(item[2])
        if conf < 0.5 or not txt.strip():
            continue
        xs = [float(p[0]) for p in box]; ys = [float(p[1]) for p in box]
        x0, x1 = min(xs), max(xs)
        y0, y1 = min(ys), max(ys)
        h = (y1 - y0) * scale
        w = (x1 - x0) * scale
        size = max(4.0, h * 0.82)
        # Historical private commentary omitted; executable code retained.
        maxw = min(w * 1.4, PAGE_W - x0 * scale - 4)
        if maxw > 0:
            est = maxw / max(1, len(txt)) / 1.02
            size = max(3.0, min(size, est))
        xs_pt = x0 * scale
        w_est = size * len(txt) * 1.02
        if xs_pt + w_est > PAGE_W - 2:            # Historical private commentary omitted; executable code retained.
            xs_pt = max(0.0, PAGE_W - 2 - w_est)
        page.insert_text((xs_pt, y1 * scale - h * 0.12), txt,
                         fontname="china-s", fontsize=size, color=(0, 0, 0),
                         render_mode=3)      # Historical private commentary omitted; executable code retained.
        n += 1

    doc.set_metadata({"title": base, "producer": "RapidOCR text layer"})
    doc.save(out_pdf, garbage=4, deflate=True)
    doc.close()
    print(f"{base:<20} {iw}x{ih}px ({iw/(PAGE_W/72):.0f}dpi) 文字层 {n} 行  {os.path.getsize(out_pdf)/1024:.0f}KB")

print("\n完成")
